Official Python MCP Client implementation: 

import asyncio
from mcp import ClientSession
from mcp.client.streamable_http import streamable_http_client

async def main():
    async with streamable_http_client("https://mcp.arif-fazil.com/mcp") as (read, write):
        async with ClientSession(read, write) as session:
            await session.initialize()
            tools = await session.list_tools()
            print(f"Connected! Loaded {len(tools.tools)} canonical tools.")

asyncio.run(main())
